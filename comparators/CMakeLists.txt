# Third-party exact predicates, as comparators. See docs/comparators.md.
#
# Opt-in (-DSHEWCHUK_BUILD_COMPARATORS=ON): this directory downloads source
# code at configure time, and CGAL needs Boost, GMP and MPFR. Nothing of the
# comparators is in this repository; each is fetched at a pinned commit or
# release, verified by its SHA-256, and compiled into one wrapper
# (comparators/<name>.cpp) with the floating-point flags its upstream
# prescribes. A comparator that cannot be built here is skipped with a
# message, and the harness reports it as unavailable.
#
# shewchuk_comparators is the interface target the test support library links.

add_library(shewchuk_comparators INTERFACE)
set(comparator_info "// Generated by comparators/CMakeLists.txt.\n")

# The architecture being compiled for (as in tests/CMakeLists.txt).
set(target_arch "${CMAKE_SYSTEM_PROCESSOR}")
if(CMAKE_OSX_ARCHITECTURES)
  set(target_arch "${CMAKE_OSX_ARCHITECTURES}")
endif()

# shewchuk_fetch_comparator_source(<name> <url> <sha256>)
#
# Downloads and unpacks an archive (nothing in it is configured or run). Sets
# <name>_SOURCE_DIR to its top-level directory, or to "" if the download
# failed: the comparator is then skipped. An archive whose SHA-256 is not the
# pinned one stops the configuration.
function(shewchuk_fetch_comparator_source name url sha256)
  set(directory "${CMAKE_BINARY_DIR}/_deps/${name}")
  set(stamp "${directory}/sha256-${sha256}")
  set(${name}_SOURCE_DIR "${directory}/source" PARENT_SCOPE)
  if(EXISTS "${stamp}")
    return()
  endif()
  set(archive "${directory}/archive")
  message(STATUS "Downloading ${url}")
  file(DOWNLOAD "${url}" "${archive}" STATUS status TLS_VERIFY ON)
  list(GET status 0 status_code)
  if(NOT status_code EQUAL 0)
    message(STATUS "Download failed: ${status}")
    file(REMOVE "${archive}")
    set(${name}_SOURCE_DIR "" PARENT_SCOPE)
    return()
  endif()
  file(SHA256 "${archive}" actual)
  if(NOT actual STREQUAL sha256)
    message(FATAL_ERROR "${url}\n  has SHA-256 ${actual}\n  expected    ${sha256}\n"
      "The pinned comparator source changed upstream; refusing to build something else.")
  endif()
  file(REMOVE_RECURSE "${directory}/unpacked" "${directory}/source")
  file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${directory}/unpacked")
  file(GLOB top_level "${directory}/unpacked/*")
  list(LENGTH top_level top_level_count)
  if(NOT top_level_count EQUAL 1)
    message(FATAL_ERROR "${url}: expected one top-level directory in the archive")
  endif()
  file(RENAME "${top_level}" "${directory}/source")
  file(REMOVE_RECURSE "${directory}/unpacked" "${archive}")
  file(WRITE "${stamp}" "${url}\n")
endfunction()

# shewchuk_add_comparator(<name> VERSION <text> FLAGS <flag>...)
#
# The wrapper comparators/<name>.cpp as the static library
# shewchuk_comparator_<name>, compiled with exactly the given floating-point
# flags after the build type's optimization flags, and registered with the
# harness. VERSION and FLAGS are also what the harness prints and records.
function(shewchuk_add_comparator name)
  cmake_parse_arguments(arg "" "VERSION" "FLAGS" ${ARGN})
  set(target shewchuk_comparator_${name})
  add_library(${target} STATIC ${name}.cpp)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(${target} PRIVATE ${arg_FLAGS})
  if(MSVC)
    target_compile_options(${target} PRIVATE /w)
  else()
    target_compile_options(${target} PRIVATE -w)  # not this project's code
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_compile_options(${target} PRIVATE -Wno-psabi)  # notes that -w leaves on
    endif()
  endif()
  target_link_libraries(shewchuk_comparators INTERFACE ${target})
  string(TOUPPER "${name}" upper)
  string(REPLACE ";" " " flags "${arg_FLAGS}")
  if(NOT flags)
    set(flags "no floating-point flag")
  endif()
  string(APPEND comparator_info
    "#define SHEWCHUK_HAVE_COMPARATOR_${upper} 1\n"
    "#define SHEWCHUK_COMPARATOR_${upper}_BUILD R\"info(${arg_VERSION}; built with ${flags})info\"\n")
  set(comparator_info "${comparator_info}" PARENT_SCOPE)
  message(STATUS "Comparator ${name}: ${arg_VERSION}; flags: ${flags}")
endfunction()

# The flag every expansion-based comparator needs: no fused multiply-add may
# replace a rounded product.
if(MSVC)
  set(no_contraction /fp:strict)
  set(rounding_math "")
else()
  set(no_contraction -ffp-contract=off)
  set(rounding_math -frounding-math)
endif()

# --- Geogram PCK (Predicates_psm) -------------------------------------------
# Flags from the CMakeLists.txt of the same archive: -ffp-contract=off
# (MSVC: /fp:strict), GEO_STATIC_LIBS, C++17 or newer, threads.
option(SHEWCHUK_COMPARATOR_GEOGRAM "Build the Geogram PCK comparator" ON)
if(SHEWCHUK_COMPARATOR_GEOGRAM)
  set(geogram_commit 8468ac11314dd4356a31dbf8b5f59ae29980e7e6)
  shewchuk_fetch_comparator_source(shewchuk_geogram_psm
    "https://github.com/BrunoLevy/geogram.psm.Predicates/archive/${geogram_commit}.tar.gz"
    58044cafffcb546641db228344c815ab50180671b7665c6f1c627699d618f4fd)
  find_package(Threads)
  if(Threads_FOUND AND EXISTS "${shewchuk_geogram_psm_SOURCE_DIR}/Predicates_psm.cpp")
    shewchuk_add_comparator(geogram
      VERSION "geogram.psm.Predicates ${geogram_commit}"
      FLAGS ${no_contraction})
    target_include_directories(shewchuk_comparator_geogram SYSTEM PRIVATE
      "${shewchuk_geogram_psm_SOURCE_DIR}")
    target_compile_definitions(shewchuk_comparator_geogram PRIVATE GEO_STATIC_LIBS)
    target_link_libraries(shewchuk_comparator_geogram PRIVATE Threads::Threads)
  else()
    message(STATUS "Comparator geogram: SKIPPED (threads or the download are missing)")
  endif()
endif()

# --- CGAL, Exact_predicates_inexact_constructions_kernel --------------------
# Header-only. Flags from cmake/modules/CGAL_SetupCGALDependencies.cmake of
# the same release: -frounding-math for GCC, /fp:strict for MSVC, nothing for
# clang; GMP and MPFR for the exact number types; Boost headers.
option(SHEWCHUK_COMPARATOR_CGAL "Build the CGAL comparator (needs Boost, GMP, MPFR)" ON)
if(SHEWCHUK_COMPARATOR_CGAL)
  set(cgal_version 6.2.1)
  find_package(GMP)
  find_path(SHEWCHUK_MPFR_INCLUDE_DIR mpfr.h HINTS /opt/homebrew/include /usr/local/include)
  find_library(SHEWCHUK_MPFR_LIBRARY mpfr HINTS /opt/homebrew/lib /usr/local/lib)
  find_path(SHEWCHUK_BOOST_INCLUDE_DIR boost/version.hpp
    HINTS /opt/homebrew/include /opt/homebrew/opt/boost/include
      /usr/local/include /usr/local/opt/boost/include)
  set(cgal_missing "")
  if(NOT GMP_FOUND)
    list(APPEND cgal_missing GMP)
  endif()
  if(NOT SHEWCHUK_MPFR_INCLUDE_DIR OR NOT SHEWCHUK_MPFR_LIBRARY)
    list(APPEND cgal_missing MPFR)
  endif()
  if(NOT SHEWCHUK_BOOST_INCLUDE_DIR)
    list(APPEND cgal_missing "Boost headers")
  endif()
  set(shewchuk_cgal_SOURCE_DIR "")
  if(NOT cgal_missing)
    shewchuk_fetch_comparator_source(shewchuk_cgal
      "https://github.com/CGAL/cgal/releases/download/v${cgal_version}/CGAL-${cgal_version}-library.tar.xz"
      8e7ce1e30e8ffcb3e69d153c3a561bb1f38bbb4becd8e6544cbaa0fb2889af62)
  endif()
  if(shewchuk_cgal_SOURCE_DIR AND EXISTS "${shewchuk_cgal_SOURCE_DIR}/include/CGAL/version.h")
    set(cgal_flags "")
    if(MSVC)
      set(cgal_flags /fp:strict /fp:except- /bigobj)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      set(cgal_flags -frounding-math)
    endif()
    shewchuk_add_comparator(cgal VERSION "CGAL ${cgal_version}" FLAGS ${cgal_flags})
    target_include_directories(shewchuk_comparator_cgal SYSTEM PRIVATE
      "${shewchuk_cgal_SOURCE_DIR}/include" "${SHEWCHUK_BOOST_INCLUDE_DIR}"
      "${SHEWCHUK_MPFR_INCLUDE_DIR}")
    target_link_libraries(shewchuk_comparator_cgal PRIVATE "${SHEWCHUK_MPFR_LIBRARY}" GMP::GMP)
  else()
    if(NOT cgal_missing)
      set(cgal_missing "the download")
    endif()
    string(REPLACE ";" ", " cgal_missing "${cgal_missing}")
    message(STATUS "Comparator cgal: SKIPPED (not found: ${cgal_missing})")
  endif()
endif()

# --- Indirect_Predicates with its numeric kernel NFG -------------------------
# Header-only. Flags from the README and CMakeLists.txt of the same commit:
# -frounding-math -ffp-contract=off (MSVC: /fp:strict), C++20, and the
# instruction set: -mavx2 -mfma on x86-64 (upstream's default; the harness
# reports the comparator as unavailable on a CPU without them). On arm64 NFG
# includes SIMDe, which upstream fetches at v0.8.2.
option(SHEWCHUK_COMPARATOR_INDIRECT "Build the Indirect_Predicates comparator" ON)
if(SHEWCHUK_COMPARATOR_INDIRECT)
  set(indirect_commit f8f243d18a6851fe3128b7e09af5b3574259d700)
  set(nfg_commit aeec97547a67dc6904ab84b39b15d20c6d62f65a)
  set(simde_commit 71fd833d9666141edcd1d3c109a80e228303d8d7)  # tag v0.8.2
  shewchuk_fetch_comparator_source(shewchuk_indirect_predicates
    "https://github.com/MarcoAttene/Indirect_Predicates/archive/${indirect_commit}.tar.gz"
    16bfd284e38fcf2fb7e3649ab5c1f704e7565278644e39732ae6043deb5fe776)
  shewchuk_fetch_comparator_source(shewchuk_nfg
    "https://github.com/MarcoAttene/nfg/archive/${nfg_commit}.tar.gz"
    0602f367fcd37c7b9256f78684c05dd9f998799fa8bbeb30d07f5a3177714690)
  set(indirect_flags ${rounding_math} ${no_contraction})
  set(indirect_includes "${shewchuk_indirect_predicates_SOURCE_DIR}/include"
    "${shewchuk_nfg_SOURCE_DIR}/include")
  set(indirect_version "Indirect_Predicates ${indirect_commit}, NFG ${nfg_commit}")
  set(indirect_ok ON)
  if(target_arch MATCHES "^(aarch64|arm64|ARM64)$")
    shewchuk_fetch_comparator_source(shewchuk_simde
      "https://github.com/simd-everywhere/simde/archive/${simde_commit}.tar.gz"
      72b2c14a487560b7eb203795f2c2fead5c7499662e639944cca2a9bb19f09029)
    list(APPEND indirect_includes "${shewchuk_simde_SOURCE_DIR}/simde")
    string(APPEND indirect_version ", SIMDe 0.8.2")
  elseif(target_arch MATCHES "^(x86_64|AMD64|amd64)$")
    if(MSVC)
      list(APPEND indirect_flags /arch:AVX2)
    else()
      list(APPEND indirect_flags -mavx2 -mfma)
    endif()
    target_compile_definitions(shewchuk_comparators INTERFACE
      SHEWCHUK_COMPARATOR_INDIRECT_USES_AVX2=1)
  else()
    set(indirect_ok OFF)
  endif()
  if(indirect_ok AND EXISTS "${shewchuk_indirect_predicates_SOURCE_DIR}/include/implicit_point.h"
      AND EXISTS "${shewchuk_nfg_SOURCE_DIR}/include/numerics.h")
    shewchuk_add_comparator(indirect VERSION "${indirect_version}" FLAGS ${indirect_flags})
    target_include_directories(shewchuk_comparator_indirect SYSTEM PRIVATE ${indirect_includes})
  else()
    message(STATUS "Comparator indirect: SKIPPED (the download is missing, or the target is "
      "neither x86-64 nor arm64)")
  endif()
endif()
