# Benchmark harness. See docs/benchmarking.md.
#
# No external dependency. The implementations come from the same table the
# exactness tests use (tests/implementations.cpp) and the inputs from the same
# generators, so the harness needs the test support library.

# --- Build description compiled into the harness and written to every result
set(build_info_dir "${CMAKE_CURRENT_BINARY_DIR}/build_info/$<CONFIG>")
get_target_property(strict_fp_flags shewchuk_strict_fp INTERFACE_COMPILE_OPTIONS)
if(NOT strict_fp_flags)
  set(strict_fp_flags "")
endif()
string(REPLACE ";" " " strict_fp_flags "${strict_fp_flags}")
set(c_flags "${CMAKE_C_FLAGS}")
set(cxx_flags "${CMAKE_CXX_FLAGS}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" config_upper)
  string(APPEND c_flags "$<$<CONFIG:${config}>: ${CMAKE_C_FLAGS_${config_upper}}>")
  string(APPEND cxx_flags "$<$<CONFIG:${config}>: ${CMAKE_CXX_FLAGS_${config_upper}}>")
endforeach()
file(GENERATE OUTPUT "${build_info_dir}/bench_build_info.h" CONTENT
"// Generated by bench/CMakeLists.txt.
#define SHEWCHUK_BENCH_BUILD_TYPE R\"info($<CONFIG>)info\"
#define SHEWCHUK_BENCH_C_COMPILER R\"info(${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION})info\"
#define SHEWCHUK_BENCH_CXX_COMPILER R\"info(${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION})info\"
#define SHEWCHUK_BENCH_C_FLAGS R\"info(${c_flags})info\"
#define SHEWCHUK_BENCH_CXX_FLAGS R\"info(${cxx_flags})info\"
#define SHEWCHUK_BENCH_STRICT_FP_FLAGS R\"info(${strict_fp_flags})info\"
")

# --- Git revision, refreshed at every build (rewritten only when it changes)
set(SHEWCHUK_BENCH_GIT_REVISION "" CACHE STRING
  "Revision recorded in benchmark results (default: asked from git at build time)")
set(git_revision_dir "${CMAKE_CURRENT_BINARY_DIR}/git_revision")
add_custom_target(shewchuk_bench_git_revision
  COMMAND "${CMAKE_COMMAND}"
    "-DSOURCE_DIR=${PROJECT_SOURCE_DIR}"
    "-DOUTPUT=${git_revision_dir}/bench_git_revision.h"
    "-DOVERRIDE=${SHEWCHUK_BENCH_GIT_REVISION}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/GitRevision.cmake"
  BYPRODUCTS "${git_revision_dir}/bench_git_revision.h"
  COMMENT "Recording the git revision for benchmark results"
  VERBATIM)

# --- The harness
add_executable(bench_predicates
  bench_predicates.cpp counters.cpp kernel.cpp overhead.cpp workloads.cpp)
target_include_directories(bench_predicates PRIVATE "${build_info_dir}" "${git_revision_dir}")
target_link_libraries(bench_predicates PRIVATE shewchuk_test_support)
add_dependencies(bench_predicates shewchuk_bench_git_revision)
if(MSVC)
  # insphereexact() alone needs about 180 KB of stack.
  target_link_options(bench_predicates PRIVATE /STACK:8388608)
  # fopen() is fine here.
  target_compile_definitions(bench_predicates PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

# --- Smoke tests: the harness and the comparison tool run. No performance is
# judged here; hosted CI machines have no usable counters and are noisy.
add_test(NAME bench.smoke COMMAND bench_predicates --smoke
  --out "${CMAKE_CURRENT_BINARY_DIR}/smoke.json")
add_test(NAME bench.smoke_wall_clock COMMAND bench_predicates --smoke --wall-clock
  --pred orient2d,insphere --out "${CMAKE_CURRENT_BINARY_DIR}/smoke_wall_clock.json")
set_tests_properties(bench.smoke bench.smoke_wall_clock PROPERTIES TIMEOUT 600)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME bench.compare_smoke COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tools/compare_revs.py"
    --binary-a "$<TARGET_FILE:bench_predicates>" --binary-b "$<TARGET_FILE:bench_predicates>"
    --rounds 6 --scale 0.002 --pred orient2d,incircle
    --json "${CMAKE_CURRENT_BINARY_DIR}/compare_smoke.json")
  add_test(NAME bench.suite_smoke COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tools/bench_suite.py"
    --binary "$<TARGET_FILE:bench_predicates>" --runs 3 --scale 0.002 --allow-invalid
    --out "${CMAKE_CURRENT_BINARY_DIR}/suite_smoke.json")
  set_tests_properties(bench.compare_smoke bench.suite_smoke PROPERTIES TIMEOUT 600)
  add_test(NAME bench.readme_table COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tools/readme_table.py" --check)
  add_test(NAME bench.comparator_table COMMAND "${Python3_EXECUTABLE}"
    "${PROJECT_SOURCE_DIR}/tools/comparator_table.py" --check)
  set_tests_properties(bench.readme_table bench.comparator_table PROPERTIES TIMEOUT 60)
else()
  message(STATUS "Python 3 not found: bench.compare_smoke and bench.suite_smoke are not registered")
endif()
